let currentFilter = "Tudo";

function renderPosts(filter) {
  const grid = document.getElementById("postsGrid");
  const filtered = filter === "Tudo" ? posts : posts.filter(p => p.tags.some(t => t.toLowerCase().includes(filter.toLowerCase())));
  grid.innerHTML = filtered.map(p => `
    <div class="card" onclick="showArticle(${p.id})">
      <div class="card-thumb" style="background:${p.bg};">${p.icon}</div>
      <div class="card-body">
        <div class="card-badge-row"><span class="badge badge-${p.badge}">${p.badgeLabel}</span></div>
        <div class="card-title">${p.shortTitle}</div>
        <div class="card-desc">${p.desc}</div>
      </div>
      <div class="card-foot">
        <span class="card-date">${p.date}</span>
        <span class="card-read">Ler →</span>
      </div>
    </div>
  `).join("");
}

function renderTrending() {
  const list = document.getElementById("trendingList");
  const trending = [posts[5], posts[1], posts[3], posts[4]];
  list.innerHTML = trending.map((p, i) => `
    <div class="trending-item" onclick="showArticle(${p.id})">
      <div class="trending-num">0${i+1}</div>
      <div class="trending-info">
        <div class="trending-ttl">${p.title}</div>
        <div class="trending-sub"><span class="badge badge-${p.badge}" style="font-size:9px;">${p.badgeLabel}</span>&nbsp;${p.read} leitura</div>
      </div>
    </div>
  `).join("");
}

function filterTag(tag) {
  currentFilter = tag;
  document.querySelectorAll(".tag").forEach(t => {
    t.classList.toggle("active", t.textContent.trim() === tag || (tag === "Tudo" && t.textContent.trim() === "Tudo"));
  });
  renderPosts(tag);
}

function showPage(name) {
  document.querySelectorAll(".page").forEach(p => p.classList.remove("active"));
  document.getElementById("page-" + name).classList.add("active");
  window.scrollTo({ top: 0, behavior: "smooth" });
}

function showArticle(id) {
  const p = posts.find(x => x.id === id);
  if (!p) return;
  let html = `
    <div class="article-hero">
      <div class="article-category">${p.category}</div>
      <h1 class="article-h1">${p.title.replace(/:/,':<br><em>').replace('</em>','</em>')}</h1>
      <div class="article-byline">
        <div class="article-author">
          <div class="author-avatar">${p.initials}</div>
          <span>${p.author}</span>
        </div>
        <span>${p.date} · ${p.read} leitura</span>
        ${p.rating ? `<div class="article-rating">★ ${p.rating}</div>` : ""}
        <span class="badge badge-${p.badge}">${p.badgeLabel}</span>
      </div>
    </div>
    <div class="article-banner" style="background:${p.bg};">${p.icon}</div>
    <div class="article-body">
  `;
  p.content.forEach(block => {
    if (block.type === "p") html += `<p>${block.text}</p>`;
    else if (block.type === "h2") html += `<h2>${block.text}</h2>`;
    else if (block.type === "quote") html += `<blockquote><p>${block.text}</p></blockquote>`;
    else if (block.type === "score") html += `
      <div class="score-box">
        <div class="score-num">${block.rating}</div>
        <div>
          <div class="score-label">Nota do Nível MAX</div>
          <div class="score-verdict">${block.verdict}</div>
        </div>
      </div>`;
    else if (block.type === "proscons") html += `
      <div class="pros-cons">
        <div class="pros">
          <div class="pros-title">✔ Prós</div>
          <ul>${block.pros.map(x => `<li>${x}</li>`).join("")}</ul>
        </div>
        <div class="cons">
          <div class="cons-title">✖ Contras</div>
          <ul>${block.cons.map(x => `<li>${x}</li>`).join("")}</ul>
        </div>
      </div>`;
  });
  html += `</div>`;

  const related = posts.filter(x => x.id !== id && x.tags.some(t => p.tags.includes(t))).slice(0,3);
  if (related.length) {
    html += `<div class="related-section">
      <div class="section-title" style="padding:0 0 1rem;">Leia também</div>
      <div class="grid">
        ${related.map(r => `
          <div class="card" onclick="showArticle(${r.id})">
            <div class="card-thumb" style="background:${r.bg};">${r.icon}</div>
            <div class="card-body">
              <div class="card-badge-row"><span class="badge badge-${r.badge}">${r.badgeLabel}</span></div>
              <div class="card-title">${r.shortTitle}</div>
            </div>
            <div class="card-foot">
              <span class="card-date">${r.date}</span>
              <span class="card-read">Ler →</span>
            </div>
          </div>`).join("")}
      </div>
    </div>`;
  }

  html += `<footer><div class="footer-logo">Nível <span>MAX</span></div><div class="footer-copy">© 2026 Nível MAX</div></footer>`;

  document.getElementById("articleContent").innerHTML = html;
  showPage("article");
}

function openSearch() {
  document.getElementById("searchOverlay").classList.add("active");
  setTimeout(() => document.getElementById("searchInput").focus(), 100);
}

function closeSearch() {
  document.getElementById("searchOverlay").classList.remove("active");
  document.getElementById("searchInput").value = "";
  document.getElementById("searchResults").innerHTML = '<div class="search-empty">Digite para buscar...</div>';
}

function doSearch(q) {
  const res = document.getElementById("searchResults");
  if (!q.trim()) { res.innerHTML = '<div class="search-empty">Digite para buscar...</div>'; return; }
  const matches = posts.filter(p =>
    p.title.toLowerCase().includes(q.toLowerCase()) ||
    p.desc.toLowerCase().includes(q.toLowerCase()) ||
    p.tags.some(t => t.toLowerCase().includes(q.toLowerCase()))
  );
  if (!matches.length) { res.innerHTML = '<div class="search-empty">Nenhum resultado encontrado.</div>'; return; }
  res.innerHTML = matches.map(p => `
    <div class="search-result-item" onclick="closeSearch();showArticle(${p.id})">
      <span style="font-size:28px;">${p.icon}</span>
      <div>
        <div class="search-result-title">${p.title}</div>
        <div class="search-result-sub"><span class="badge badge-${p.badge}" style="margin-right:6px;">${p.badgeLabel}</span>${p.date} · ${p.read}</div>
      </div>
    </div>
  `).join("");
}

document.addEventListener("keydown", e => { if (e.key === "Escape") closeSearch(); });
